# MSEB / SVQ benchmark (acoustic-hypothesis reranking + audio→text retrieval).
#
# This lives outside pyproject.toml on purpose. `mseb` requires Python >= 3.12,
# while this project supports >= 3.10, so it cannot be an extra without either
# raising the floor for everyone or making the lockfile unresolvable. The
# workflow docs/MSEB.md already prescribes a dedicated 3.12 virtualenv:
#
#     uv venv --python 3.12 .venv-mseb
#     source .venv-mseb/bin/activate
#     uv pip install -r requirements-mseb.txt -e .
#
# (The trailing `-e .` installs paraspeechrag itself into that env.)

mseb @ git+https://github.com/google-research/mseb.git

# Pinned by name, not version: the reranking evaluator imports
# `whisper.normalizers`, which the similarly-named PyPI `whisper` package does
# not provide. Installing the wrong one fails at import.
openai-whisper

sentence-transformers
torchvision
Pillow

# Floor apache-beam to a Python 3.12-compatible release; without this the
# resolver can backtrack to an ancient 2.3.0 that fails to build on 3.12.
apache-beam>=2.60
